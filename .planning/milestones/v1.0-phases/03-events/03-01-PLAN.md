    ```dart
    /// All non-deleted events in a group, sorted: no-date events first, then by createdAt desc.
    final groupEventsProvider = StreamProvider.family<List<Event>, String>((ref, groupId) {
      return FirebaseConfig.firestore
          .collection('groups')
          .doc(groupId)
          .collection('events')
          .where('isDeleted', isEqualTo: false)
          .orderBy('createdAt', descending: true)
          .snapshots()
          .map((snap) {
            final events = snap.docs.map(Event.fromDoc).toList();
            events.sort((a, b) {
              if (a.startDate == null && b.startDate == null) {
                return b.createdAt.compareTo(a.createdAt);
              }
              if (a.startDate == null) return -1;
              if (b.startDate == null) return 1;
              return b.createdAt.compareTo(a.createdAt);
            });
            return events;
          });
    });

    /// Single event detail stream using compound key.
    final eventDetailProvider = StreamProvider.family<Event?, ({String groupId, String eventId})>(
        (ref, params) {
      return FirebaseConfig.firestore
          .collection('groups')
          .doc(params.groupId)
          .collection('events')
          .doc(params.eventId)
          .snapshots()
          .map((doc) => doc.exists ? Event.fromDoc(doc) : null);
    });
    ```

    **1. Update security/firestore.rules:**

    Add a specific `match /events/{eventId}` rule INSIDE the `match /groups/{groupId}` block, BEFORE the generic `match /{subcollection}/{docId}` rule. The specific rule shadows the generic one for the events path (Firestore applies most specific match).

    Add this block between the group-level rules and the generic subcollection rule:

    ```
    // Events subcollection -- more restrictive than generic subcollection rule (D-32)
    match /events/{eventId} {
      function isGroupMemberForEvent() {
        return request.auth != null &&
          request.auth.uid in
            get(/databases/$(database)/documents/groups/$(groupId)).data.memberIds;
      }

      function isEventParticipant() {
        return request.auth != null &&
          request.auth.uid in resource.data.participantIds;
      }

      function isValidEventCreate() {
        return request.auth != null &&
          request.resource.data.name is string &&
          request.resource.data.name.size() > 0 &&
          request.resource.data.type is string &&
          request.resource.data.participantIds is list &&
          request.resource.data.participantIds.size() > 0;
      }

      // Any group member can read all events (transparency, D-32)
      allow read: if isGroupMemberForEvent();
      // Any group member can create events (D-07)
      allow create: if isGroupMemberForEvent() && isValidEventCreate();
      // Only event participants can update event data
      allow update: if isEventParticipant();
      // No hard deletes -- soft delete only via update (D-09)
      allow delete: if false;
    }
    ```

    The existing generic `match /{subcollection}/{docId}` rule remains unchanged -- it still covers `members` and other subcollections. The `events` path now uses the more specific rule.
