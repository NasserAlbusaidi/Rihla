    ```dart
    import 'package:cloud_firestore/cloud_firestore.dart';
    import 'package:flutter/foundation.dart';
    import '../config/firebase_config.dart';

    abstract class FirestoreRepository {
      final FirebaseFirestore _db;

      /// Production constructor -- uses FirebaseConfig singleton.
      FirestoreRepository() : _db = FirebaseConfig.firestore;

      /// Test constructor -- injects FakeFirebaseFirestore.
      @visibleForTesting
      FirestoreRepository.withFirestore(FirebaseFirestore db) : _db = db;

      FirebaseFirestore get db => _db;

      /// Convenience: events subcollection reference.
      CollectionReference<Map<String, dynamic>> eventSubcollection(
        String groupId,
        String eventId,
        String module,
      ) {
        return _db
            .collection('groups')
            .doc(groupId)
            .collection('events')
            .doc(eventId)
            .collection(module);
      }
    }
    ```

    Replace the existing generic group subcollection rule with a more specific rule for event module subcollections. The current rules have:
    - `match /groups/{groupId}` with group-level rules
    - `match /events/{eventId}` nested under groups with event-level rules
    - `match /{subcollection}/{docId}` generic catch-all for group subcollections

    Add a NEW rule nested under the events match block, BEFORE the generic subcollection catch-all:

    ```
    // Module subcollections (expenses, settlements, gear, etc.)
    // Any event participant can read/write module data (D-11)
    match /events/{eventId}/{module}/{docId} {
      function isEventParticipantForModule() {
        return request.auth != null &&
          request.auth.uid in
            get(/databases/$(database)/documents/groups/$(groupId)/events/$(eventId)).data.participantIds;
      }
      allow read, write: if isEventParticipantForModule();
    }
    ```

    This rule goes inside the `match /groups/{groupId}` block, at the same level as the existing `match /events/{eventId}` block. It uses ONE get() call to the event document to check participantIds -- this is within the 10-get limit per Pitfall 3.
