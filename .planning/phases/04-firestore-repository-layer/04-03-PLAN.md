    ```dart
    class DocumentService extends FirestoreRepository {
      DocumentService() : super();

      @visibleForTesting
      DocumentService.withFirestore(FirebaseFirestore db) : super.withFirestore(db);

      FirebaseStorage get _storage => FirebaseStorage.instance;

      static const int maxFileSizeBytes = 25 * 1024 * 1024; // 25 MB

      Stream<List<Document>> watchDocuments(String groupId, String eventId) {
        return eventSubcollection(groupId, eventId, 'documents')
            .orderBy('createdAt', descending: true)
            .snapshots()
            .map((snap) => snap.docs
                .map((doc) => Document.fromFirestore({...doc.data(), 'id': doc.id}))
                .toList());
      }

      Future<Document> uploadFile({
        required String groupId,
        required String eventId,
        required String filePath,
        required String fileName,
        required int fileSize,
        String? mimeType,
      }) async {
        final uid = FirebaseConfig.currentUser?.uid;
        if (uid == null) throw Exception('Not authenticated');

        final id = const Uuid().v4();
        final timestamp = DateTime.now().millisecondsSinceEpoch;
        final storagePath = 'trip-documents/$eventId/$timestamp-$fileName';

        // Upload to Firebase Storage
        final ref = _storage.ref().child(storagePath);
        final metadata = SettableMetadata(contentType: mimeType);
        await ref.putFile(File(filePath), metadata);

        // Write metadata to Firestore
        final now = DateTime.now().toUtc();
        final data = {
          'id': id,
          'eventId': eventId,
          'uploaderId': uid,
          'storagePath': storagePath,
          'fileName': fileName,
          'fileSize': fileSize,
          'mimeType': mimeType,
          'createdAt': now.toIso8601String(),
        };
        await eventSubcollection(groupId, eventId, 'documents').doc(id).set(data);
        return Document.fromFirestore(data);
      }

      Future<String?> getDownloadUrl(String storagePath) async {
        try {
          return await _storage.ref().child(storagePath).getDownloadURL();
        } catch (e) {
          return null;
        }
      }

      Future<bool> deleteDocument({
        required String groupId,
        required String eventId,
        required String documentId,
        required String storagePath,
      }) async {
        try {
          await _storage.ref().child(storagePath).delete();
          await eventSubcollection(groupId, eventId, 'documents').doc(documentId).delete();
          return true;
        } catch (e) {
          return false;
        }
      }
    }
    ```

    **Create `lib/core/services/lazy_migration_service.dart`** per D-18:

    This service is called when a user opens an event module screen (e.g., Ledger, Gear) for an event that has a `bridgeTripId`. If the Firestore subcollection for that module is empty, it fetches data from Supabase via the bridge trip ID and writes it to Firestore.

    ```dart
    import 'package:cloud_firestore/cloud_firestore.dart';
    import 'package:flutter/foundation.dart';

    import '../config/firebase_config.dart';
    import '../config/supabase_config.dart';
    import 'money_serializer.dart';

    class LazyMigrationService {
      final FirebaseFirestore _db;

      LazyMigrationService() : _db = FirebaseConfig.firestore;

      @visibleForTesting
      LazyMigrationService.withFirestore(FirebaseFirestore db) : _db = db;

      /// Check if a module's Firestore subcollection needs migration.
      /// Returns true if migration was performed, false if already migrated or skipped.
      Future<bool> migrateModuleIfNeeded({
        required String groupId,
        required String eventId,
        required String? bridgeTripId,
        required String module,
      }) async {
        if (bridgeTripId == null || bridgeTripId.isEmpty) return false;

        // Check if Firestore already has data
        final snapshot = await _db
            .collection('groups/$groupId/events/$eventId/$module')
            .limit(1)
            .get();
        if (snapshot.docs.isNotEmpty) return false; // Already migrated

        try {
          // Check if Supabase is available
          if (!SupabaseConfig.isAuthenticated) return false;

          switch (module) {
            case 'expenses':
              return await _migrateExpenses(groupId, eventId, bridgeTripId);
            case 'settlements':
              return await _migrateSettlements(groupId, eventId, bridgeTripId);
            case 'gear_items':
              return await _migrateGearItems(groupId, eventId, bridgeTripId);
            case 'sub_groups':
              return await _migrateSubGroups(groupId, eventId, bridgeTripId);
            case 'activity_logs':
              return await _migrateActivityLogs(groupId, eventId, bridgeTripId);
            default:
              return false;
          }
        } catch (e) {
          debugPrint('LazyMigration: Failed to migrate $module for event $eventId: $e');
          return false;
        }
      }
    }
    ```

    Each `_migrateX` method:
    1. Reads from Supabase table where `trip_id = bridgeTripId`
    2. Converts each row from Supabase snake_case to Firestore camelCase format
    3. For expenses/settlements: converts `amount` (String Decimal in Supabase) to `amountFils` (int) via `MoneySerializer.toSubunits(Decimal.parse(row['amount']), row['currency'] ?? 'OMR')`
    4. Writes each document to the Firestore subcollection using `WriteBatch` (same-type batch per Pitfall 4 — do NOT cross subcollections)
    5. Uses the original Supabase `id` as the Firestore document ID to maintain consistency
    6. Returns `true` on success

    **Error handling per UI-SPEC:** Migration failures are non-fatal. The service catches exceptions and returns `false`. The calling screen (module screen) interprets `false` as migration error and shows the error state from UI-SPEC (EmptyStateView with `Icons.sync_problem_rounded`, "Couldn't load your data", "Try Again" button).

    **Wrap Supabase access in try-catch** per Phase 3 decision: Supabase `isAuthenticated` check wrapped in try-catch to handle uninitialized Supabase in test/Firebase-only environments.

    **Tests:** Use FakeFirebaseFirestore for the Firestore side. Mock Supabase calls using mocktail. Test the skip-if-non-empty path, the migrate-from-Supabase path, and the error-handling path.
