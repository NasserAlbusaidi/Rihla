    ```dart
    List<GroupActivityLog> get _filteredActivities {
      if (_activeFilter == 'All') return _activities;
      return _activities.where((a) {
        return switch (_activeFilter) {
          'Payments' => a.type == 'group_settlement',
          'Events' => a.type == 'event_created' || a.type == 'event_deleted',
          'Members' => a.type == 'member_joined' || a.type == 'member_left',
          _ => true,
        };
      }).toList();
    }
    ```

    ```dart
    final filtered = _filteredActivities;
    final grouped = _groupByDate(filtered);
    final flatItems = <dynamic>[]; // Mix of String (date label) and GroupActivityLog
    for (final entry in grouped.entries) {
      flatItems.add(entry.key); // date label string
      flatItems.addAll(entry.value); // activity logs
    }
    if (_hasMore) flatItems.add('__loading__'); // loading indicator sentinel

    ListView.builder(
      controller: _scrollController,
      padding: EdgeInsets.symmetric(horizontal: 12, vertical: 8),
      itemCount: flatItems.length,
      itemBuilder: (context, index) {
        final item = flatItems[index];
        if (item == '__loading__') {
          return Center(child: Padding(
            padding: EdgeInsets.all(16),
            child: SizedBox(width: 16, height: 16, child: CircularProgressIndicator(strokeWidth: 2, color: AppColorTokens.light.primary)),
          ));

        if (item is String) {
          return _DateSectionHeader(label: item);
        }
        final activity = item as GroupActivityLog;
        final sectionLogs = grouped.values.expand((e) => e).toList();
        final localIndex = sectionLogs.indexOf(activity);
        return GroupActivityTile(activity: activity)
          .animate()
          .fadeIn(delay: Duration(milliseconds: (localIndex % 10) * 50))
          .slideY(begin: 0.1, curve: Curves.easeOutCubic);
      },
    )
    ```
