**_loadPage method:**
```dart
Future<void> _loadPage() async {
  if (_isLoadingMore || !_hasMore) return;
  setState(() => _isLoadingMore = true);

  final service = ref.read(groupActivityServiceProvider);
  final snap = await service.fetchActivityPageRaw(
    widget.groupId,
    startAfter: _lastDocument,
    limit: 50,
  );

  final newActivities = snap.docs
      .map((doc) => GroupActivityLog.fromFirestore({...doc.data(), 'id': doc.id}))
      .toList();

  setState(() {
    _activities.addAll(newActivities);
    _lastDocument = snap.docs.isNotEmpty ? snap.docs.last : null;
    _hasMore = snap.docs.length == 50;
    _isLoadingMore = false;
  });
}
```
