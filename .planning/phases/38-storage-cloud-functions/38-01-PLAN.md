```typescript
import { getFirestore } from 'firebase-admin/firestore';
import './../src/admin';

export interface SeedOptions {
  groupId: string;
  eventId: string;
  memberIds: string[];
}

export async function seedGroupWithEvent(opts: SeedOptions): Promise<void> {
  const db = getFirestore();
  await db.doc(`groups/${opts.groupId}`).set({
    id: opts.groupId,
    name: `TestGroup-${opts.groupId}`,
    memberIds: opts.memberIds,
    createdAt: new Date(),
  });
  await db.doc(`groups/${opts.groupId}/events/${opts.eventId}`).set({
    id: opts.eventId,
    groupId: opts.groupId,
    name: `TestEvent-${opts.eventId}`,
    createdAt: new Date(),
  });
}

export async function clearFirestore(): Promise<void> {
  const db = getFirestore();
  const groups = await db.collection('groups').listDocuments();
  for (const g of groups) {
    const events = await g.collection('events').listDocuments();
    for (const e of events) { await e.delete(); }
    await g.delete();
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Flutter client → Cloud Functions | Untrusted client data (groupId, eventId, fileName, contentType, sizeBytes) crosses here when callables are invoked (Wave 1+). |
| Cloud Functions → Firestore Admin SDK | Trusted: runs with service account credentials. |
| Cloud Functions → GCS | Trusted: signed URLs issued here are handed back to untrusted client. |
| Dev machine → Firebase Emulator | Trusted local loopback; no prod data. |

## STRIDE Threat Register (Wave 0 scope)

Wave 0 creates no runtime surface — threats below are latent until callables land in Wave 1. Included for traceability.

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-38-01 | Spoofing | All callables (latent — scaffold only) | mitigate | Wave 1 adds `if (!request.auth) throw HttpsError('unauthenticated')` in every callable. Scaffold exports no callables yet. |
| T-38-02 | Elevation of Privilege | Cross-group access (latent) | mitigate | Wave 1 adds `assertMemberOfEvent(uid, groupId, eventId)` helper reading `groups/{gid}.memberIds`. |
| T-38-03 | Tampering | Oversized upload (latent) | mitigate | Wave 1 signing helper sets `x-goog-content-length-range: 0,26214400` on signed URL. |
| T-38-04 | EoP | Direct Storage SDK bypass | mitigate | Wave 3 tightens `security/storage.rules` to deny-all for trip-documents/*, trip-memories/*, receipts/*. Wave 0 leaves rules untouched. |
| T-38-05 | Information Disclosure | Signed URL leak | mitigate | Wave 1 signing helper pins 15min upload / 60min download TTLs (per D-02). |
| T-38-06 | EoP | Malformed storagePath on delete (latent) | mitigate | Wave 2 `deleteStorageObject` callable will parse storagePath, re-derive eventId, re-check membership. |
| T-38-07 | Tampering | Build-time config leak (`USE_FIREBASE_EMULATOR` pointing prod to localhost) | mitigate | Flag defaults `false` in config.json.example; release builds cannot accidentally use emulator because `const bool.fromEnvironment` bakes at compile time. |
| T-38-08 | Spoofing | Blaze plan not enabled → deploy fails silently mid-rollout | accept (operational) | Task 3 checkpoint verifies Blaze BEFORE Wave 3 deploy runs. |
</threat_model>
