class ExpenseCacheRepository {
  Future<void> cacheExpenses(String eventId, List<Expense> expenses) async {
    final db = await LocalDatabase.database;
    // NOTE: 'trip_id' column stores eventId — see local_database.dart for schema.
    await db.delete('expenses', where: 'trip_id = ?', whereArgs: [eventId]);
    final batch = db.batch();
    for (final e in expenses) {
      batch.insert('expenses', e.toSqliteMap(), conflictAlgorithm: ConflictAlgorithm.replace);
    }
    await batch.commit(noResult: true);
  }

  Future<List<Expense>> getExpenses(String eventId) async {
    final db = await LocalDatabase.database;
    final rows = await db.query('expenses', where: 'trip_id = ?', whereArgs: [eventId]);
    return rows.map(Expense.fromSqliteMap).toList();
  }
}
