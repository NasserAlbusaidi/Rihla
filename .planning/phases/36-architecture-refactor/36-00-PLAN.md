  ```dart
  import 'package:flutter_test/flutter_test.dart';

  /// ARCH-01: No screen file in lib/features/**/screens/ exceeds 600 lines.
  ///
  /// This test is the source-of-truth finish line for Phase 36 screen decomposition.
  /// Fails RED today with 5 violations. Turns GREEN when Wave 1 plans ship.
  void main() {
    group('ARCH-01: Screen LOC ceiling', () {
      test('no screen file exceeds 600 lines', () async {
        final dir = Directory('lib/features');
        expect(dir.existsSync(), isTrue, reason: 'lib/features must exist');

        final screens = await dir
            .list(recursive: true)
            .where((e) => e is File)
            .where((e) => e.path.contains('/screens/'))
            .where((e) => e.path.endsWith('.dart'))
            .cast<File>()
            .toList();

        expect(screens, isNotEmpty, reason: 'Expected at least one screen');

        final violations = <String>[];
        for (final file in screens) {
          final lineCount = (await file.readAsLines()).length;
          if (lineCount > 600) {
            violations.add('${file.path}: $lineCount lines');
          }
        }

        expect(
          violations,
          isEmpty,
          reason:
              'ARCH-01 requires all screen files <= 600 LOC. '
              'Violations:\n${violations.join('\n')}',
        );
      });
    });
  }
  ```
