```dart
Future<void> _save() async {
  final name = _nameController.text.trim();
  if (name.isEmpty) {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text("Event name can't be empty."), duration: Duration(seconds: 2)),
    );
    return;
  }
  setState(() => _isSaving = true);
  try {
    await ref.read(eventServiceProvider).updateEvent(
      groupId: widget.event.groupId,
      eventId: widget.event.id,
      name: name != widget.event.name ? name : null,
      startDate: _startDate,
      endDate: _endDate,
      description: _descriptionController.text.trim().isNotEmpty
          ? _descriptionController.text.trim()
          : null,
    );
    if (mounted) {
      HapticService.success();
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: const Text('Event updated', style: TextStyle(color: Colors.white)),
          backgroundColor: AppColorTokens.light.textPrimary,
          behavior: SnackBarBehavior.floating,
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
          duration: const Duration(seconds: 2),
        ),
      );
    }
  } catch (e) {
    if (mounted) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: const Text("Couldn't save changes. Try again."),
          duration: const Duration(seconds: 3),
        ),
      );
    }
  } finally {
    if (mounted) setState(() => _isSaving = false);
  }
}
```
